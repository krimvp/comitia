def summarize(rows):
    result = []
    for row in rows:
        found = False
        for item in result:
            if item["user"] == row["user"]:
                item["total"] += row["amount"]
                found = True
                break
        if not found:
            result.append({"user": row["user"], "total": row["amount"]})
    return result
